# Servicios systemd individuales
## Miembro A (eva) - /etc/systemd/system/structure-monitor.service
[director] sudo tee /etc/systemd/system/structure-monitor.service <<'EOF'
[Unit]
Description=AURORA-6 monitor de estructura (miembro A, eva)
After=network.target

[Service]
Type=simple
User=eva
ExecStart=/opt/aurora6/tools/structure-monitor.sh
Restart=on-failure
RestartSec=3

[Install]
WantedBy=multi-user.target
EOF
